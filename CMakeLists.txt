cmake_minimum_required(VERSION 3.16)
project(ROADRUNNER VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(RR_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(RR_BUILD_TESTS "Build conformance tests" ON)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
  if(RR_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
add_library(roadrunner src/roadrunner.c)
target_include_directories(roadrunner PUBLIC include)
add_executable(rr src/cli.c)
target_link_libraries(rr PRIVATE roadrunner)
add_executable(rr_bench benchmarks/bench.c)
target_link_libraries(rr_bench PRIVATE roadrunner)
add_executable(rr_plan_bench benchmarks/plan_bench.c)
target_link_libraries(rr_plan_bench PRIVATE roadrunner)
if(RR_BUILD_TESTS)
  enable_testing()
  add_executable(rr_tests tests/test_roadrunner.c)
  target_link_libraries(rr_tests PRIVATE roadrunner)
  add_test(NAME conformance COMMAND rr_tests)
  add_executable(rr_tests_wrap tests/test_roadrunner.c src/roadrunner.c)
  target_include_directories(rr_tests_wrap PRIVATE include)
  target_compile_definitions(rr_tests_wrap PRIVATE RR_EPOCH_LIMIT=3)
  add_test(NAME epoch_rollover COMMAND rr_tests_wrap)
  add_executable(rr_bench_zero_clock tests/bench_zero_clock.c)
  target_link_libraries(rr_bench_zero_clock PRIVATE roadrunner)
  add_executable(rr_plan_bench_frozen benchmarks/plan_bench.c)
  target_compile_definitions(rr_plan_bench_frozen PRIVATE RR_PLAN_FROZEN_CLOCK)
  target_link_libraries(rr_plan_bench_frozen PRIVATE roadrunner)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME plan_benchmark COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_plan_bench.py $<TARGET_FILE:rr_plan_bench> $<TARGET_FILE:rr_plan_bench_frozen>)
    add_test(NAME benchmark_oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_benchmark_oracle.py $<TARGET_FILE:rr_bench>)
    add_test(NAME benchmark_zero_clock COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_benchmark_oracle.py $<TARGET_FILE:rr_bench_zero_clock> --zero-clock)
    add_test(NAME benchmark_comparison COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_benchmark_comparison.py $<TARGET_FILE:rr_bench_zero_clock>)
    add_test(NAME cli_parser COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:rr>)
  endif()
endif()
